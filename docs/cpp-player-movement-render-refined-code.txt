// ---------- 1. 数据与常量 ----------
enum PLR_MODE { PM_STAND, PM_WALK_NORTHWARDS, PM_WALK_SOUTHWARDS, PM_WALK_SIDEWAYS, ... };
enum Direction { South, SouthWest, West, NorthWest, North, NorthEast, East, SouthEast };

const DirectionSettings WalkSettings[8] = {
    { South, PM_WALK_SOUTHWARDS }, { SouthWest, PM_WALK_SOUTHWARDS }, { West, PM_WALK_SIDEWAYS },
    { NorthWest, PM_WALK_NORTHWARDS }, { North, PM_WALK_NORTHWARDS }, { NorthEast, PM_WALK_NORTHWARDS },
    { East, PM_WALK_SIDEWAYS }, { SouthEast, PM_WALK_SOUTHWARDS }
};

const int MovingOffset[8][2] = {
    { 0, 32 }, { -32, 16 }, { -64, 0 }, { -32, -16 },
    { 0, -32 }, { 32, -16 }, { 64, 0 }, { 32, 16 }
};

int8_t dPlayer[MAXDUNX][MAXDUNY];

void occupyTile(Player* p, Point tile, bool isMoving) {
    int id = p->getId() + 1;
    dPlayer[tile.x][tile.y] = isMoving ? -id : id;
}

// ---------- 2. 开始行走 ----------
void StartWalk(Player* player, Direction dir, bool pmWillBeCalled) {
    StartWalkAnimation(player, dir, pmWillBeCalled);
    HandleWalkMode(player, dir);
}

// StartWalkAnimation 具体实现 (player.cpp:144-151)
void StartWalkAnimation(Player* player, Direction dir, bool pmWillBeCalled) {
    int8_t numSkippedFrames = -2;
    if (leveltype == DTYPE_TOWN && sgGameInitInfo.bRunInTown != 0)
        numSkippedFrames = 2;
    if (pmWillBeCalled)
        numSkippedFrames += 1;
    NewPlrAnim(player, player_graphic::Walk, dir,
               AnimationDistributionFlags::ProcessAnimationPending,
               numSkippedFrames);
}

// NewPlrAnim (player.cpp:2219-2235)：为 Walk 设定动画并启播
void NewPlrAnim(Player* player, player_graphic graphic, Direction dir,
                AnimationDistributionFlags flags, int8_t numSkippedFrames,
                int8_t distributeFramesBeforeFrame, uint8_t previewShownGameTickFragments) {
    LoadPlrGFX(player, graphic);
    OptionalClxSpriteList sprites = player->AnimationData[(size_t)graphic].spritesForDirection(dir);
    int8_t numberOfFrames, ticksPerFrame;
    player->getAnimationFramesAndTicksPerFrame(graphic, numberOfFrames, ticksPerFrame);
    // Walk: numberOfFrames = _pWFrames, ticksPerFrame = 1
    player->AnimInfo.setNewAnimation(sprites, numberOfFrames, ticksPerFrame,
                                     flags, numSkippedFrames, distributeFramesBeforeFrame,
                                     previewShownGameTickFragments);
}

// getAnimationFramesAndTicksPerFrame 对 Walk (player.cpp:1825-1827)
//   numberOfFrames = _pWFrames; ticksPerFrame 保持 1

void HandleWalkMode(Player* player, Direction dir) {
    DirectionSettings* ds = &WalkSettings[(int)dir];
    if (!PlrDirOK(player, dir)) return;
    player->_pdir = dir;
    player->position.future = player->position.tile + ds->dir;
    occupyTile(player, player->position.future, true);
    player->position.temp = player->position.tile + ds->dir;
    player->_pmode = ds->walkMode;
}

// ---------- 3. 每 tick 推进 ----------
void ProcessPlayers() {
    for (Player& player : active_players) {
        do {
            bool tplayer = false;
            switch (player._pmode) {
                case PM_WALK_NORTHWARDS:
                case PM_WALK_SOUTHWARDS:
                case PM_WALK_SIDEWAYS:
                    tplayer = DoWalk(&player);
                    break;
                default: break;
            }
            CheckNewPath(&player, tplayer);
        } while (tplayer);
        player.AnimInfo.processAnimation();
    }
}

bool DoWalk(Player* player) {
    if (!player->AnimInfo.isLastFrame()) {
        UpdatePlayerLightOffset(player);
        return false;
    }
    dPlayer[player->position.tile.x][player->position.tile.y] = 0;
    player->position.tile = player->position.temp;
    occupyTile(player, player->position.tile, false);
    StartStand(player, player->tempDirection);
    ClearStateVariables(player);
    return true;
}

void AnimationInfo::processAnimation() {
    tickCounterOfCurrentFrame++;
    ticksSinceSequenceStarted_ += baseValueFraction;
    if (tickCounterOfCurrentFrame >= ticksPerFrame) {
        tickCounterOfCurrentFrame = 0;
        currentFrame++;
        if (currentFrame >= numberOfFrames) currentFrame = 0;
    }
}

// ---------- 4. 行走进度与像素偏移 ----------
uint8_t AnimationInfo::getAnimationProgress() {
    int32_t ticks = (currentFrame * ticksPerFrame + tickCounterOfCurrentFrame) * baseValueFraction;
    int32_t total = getProgressToNextGameTick() + ticks;
    int32_t frac = total * (baseValueFraction / ticksPerFrame) / numberOfFrames / baseValueFraction;
    return (uint8_t)min(frac, baseValueFraction);
}

Displacement GetOffsetForWalking(AnimationInfo* anim, Direction dir, bool cameraMode) {
    uint8_t progress = anim->getAnimationProgress();
    Displacement off = { MovingOffset[(int)dir][0], MovingOffset[(int)dir][1] };
    off *= progress; off /= baseValueFraction;
    if (cameraMode) { off.x = -off.x; off.y = -off.y; }
    return off;
}

Displacement Player::getRenderingOffset(ClxSprite sprite) {
    Displacement off = { -CalculateSpriteTileCenterX(sprite.width()), 0 };
    if (isWalking()) off += GetOffsetForWalking(&AnimInfo, _pdir);
    return off;
}

// ---------- 5. 按格绘制玩家 ----------
void DrawCell_PlayerBranch(Surface* out, Point tilePosition, Point targetBufferPosition, int lightTableIndex) {
    Player* player = PlayerAtPosition(tilePosition);
    if (!player) return;
    int playerId = player->getId() + 1;
    if (player->_pmode == PM_WALK_SOUTHWARDS || (player->_pmode == PM_WALK_SIDEWAYS && player->_pdir == East))
        playerId = -playerId;
    if (dPlayer[tilePosition.x][tilePosition.y] != playerId) return;

    Point drawTile = tilePosition;
    Point drawBufferPos = targetBufferPosition;
    if (player->_pmode == PM_WALK_SOUTHWARDS) {
        switch (player->_pdir) {
            case SouthWest: drawBufferPos += { TILE_WIDTH/2, -TILE_HEIGHT/2 }; break;
            case South:     drawBufferPos += { 0, -TILE_HEIGHT }; break;
            case SouthEast: drawBufferPos += { -TILE_WIDTH/2, -TILE_HEIGHT/2 }; break;
            default: break;
        }
        drawTile += Opposite(player->_pdir);
    } else if (player->_pmode == PM_WALK_SIDEWAYS && player->_pdir == East) {
        drawBufferPos += { -TILE_WIDTH, 0 };
        drawTile += Opposite(player->_pdir);
    }
    DrawPlayer(out, player, drawTile, drawBufferPos, lightTableIndex);
}

void DrawPlayer(Surface* out, Player* player, Point tilePosition, Point targetBufferPosition, int lightTableIndex) {
    ClxSprite sprite = player->currentSprite();
    Point pos = targetBufferPosition + player->getRenderingOffset(sprite);
    ClxDraw(out, pos, sprite);
}
